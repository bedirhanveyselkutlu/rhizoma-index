def flatten(lst: list) -> list:
    result = []
    stack = [iter(lst)]
    while stack:
        for item in stack[-1]:
            if isinstance(item, list):
                stack.append(iter(item))
                break
            else:
                result.append(item)
        else:
            stack.pop()
    return result
