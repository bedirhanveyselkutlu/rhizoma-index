function objectDiff(obj1, obj2) {
  function isPlainObject(val) {
    return val !== null && typeof val === 'object' && !Array.isArray(val);
  }

  function deepEqual(a, b) {
    if (a === b) return true;
    if (typeof a !== typeof b) return false;
    if (a === null || b === null) return false;
    if (typeof a !== 'object') return false;
    if (Array.isArray(a) !== Array.isArray(b)) return false;
    const keysA = Object.keys(a);
    const keysB = Object.keys(b);
    if (keysA.length !== keysB.length) return false;
    return keysA.every(function (key) {
      return Object.prototype.hasOwnProperty.call(b, key) && deepEqual(a[key], b[key]);
    });
  }

  const diff = {};
  const keys = new Set([...Object.keys(obj1 || {}), ...Object.keys(obj2 || {})]);

  keys.forEach(function (key) {
    const oldVal = obj1 ? obj1[key] : undefined;
    const newVal = obj2 ? obj2[key] : undefined;

    if (isPlainObject(oldVal) && isPlainObject(newVal)) {
      const nested = objectDiff(oldVal, newVal);
      if (Object.keys(nested).length > 0) {
        diff[key] = nested;
      }
    } else if (!deepEqual(oldVal, newVal)) {
      diff[key] = [oldVal, newVal];
    }
  });

  return diff;
}

module.exports = { objectDiff };
