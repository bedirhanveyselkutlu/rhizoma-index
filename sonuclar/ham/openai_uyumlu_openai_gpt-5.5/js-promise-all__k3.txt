```javascript
function promiseAll(promises) {
  return new Promise((resolve, reject) => {
    const results = [];
    let completed = 0;
    let index = 0;

    for (const promise of promises) {
      const currentIndex = index++;
      Promise.resolve(promise)
        .then((value) => {
          results[currentIndex] = value;
          completed++;

          if (completed === index) {
            resolve(results);
          }
        })
        .catch(reject);
    }

    if (index === 0) {
      resolve([]);
    }
  });
}

module.exports = { promiseAll };
```
