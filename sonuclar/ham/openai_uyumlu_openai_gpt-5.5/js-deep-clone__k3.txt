function deepClone(obj) {
  const seen = new WeakMap();

  function clone(value) {
    if (value === null || typeof value !== "object") {
      return value;
    }

    if (seen.has(value)) {
      return seen.get(value);
    }

    let cloned;

    if (value instanceof Date) {
      cloned = new Date(value.getTime());
      seen.set(value, cloned);
    } else if (value instanceof RegExp) {
      cloned = new RegExp(value.source, value.flags);
      cloned.lastIndex = value.lastIndex;
      seen.set(value, cloned);
    } else if (Array.isArray(value)) {
      cloned = new Array(value.length);
      seen.set(value, cloned);
    } else {
      cloned = Object.create(Object.getPrototypeOf(value));
      seen.set(value, cloned);
    }

    const descriptors = Object.getOwnPropertyDescriptors(value);

    Reflect.ownKeys(descriptors).forEach((key) => {
      if ("value" in descriptors[key]) {
        descriptors[key].value = clone(descriptors[key].value);
      }
    });

    Object.defineProperties(cloned, descriptors);

    return cloned;
  }

  return clone(obj);
}

module.exports = { deepClone };
