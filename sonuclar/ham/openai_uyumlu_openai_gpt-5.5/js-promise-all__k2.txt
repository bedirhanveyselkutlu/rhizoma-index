function promiseAll(promises) {
  return new Promise((resolve, reject) => {
    let items;

    try {
      items = Array.from(promises);
    } catch (error) {
      reject(error);
      return;
    }

    const results = new Array(items.length);
    let fulfilledCount = 0;

    if (items.length === 0) {
      resolve(results);
      return;
    }

    items.forEach((promise, index) => {
      Promise.resolve(promise)
        .then((value) => {
          results[index] = value;
          fulfilledCount += 1;

          if (fulfilledCount === items.length) {
            resolve(results);
          }
        })
        .catch(reject);
    });
  });
}

module.exports = { promiseAll };
