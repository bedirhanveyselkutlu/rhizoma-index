```javascript
function throttle(fn, limit) {
  let wait = false;
  let storedArgs = null;
  let storedThis = null;

  function checkStoredArgs() {
    if (storedArgs === null) {
      wait = false;
    } else {
      fn.apply(storedThis, storedArgs);
      storedArgs = null;
      storedThis = null;
      setTimeout(checkStoredArgs, limit);
    }
  }

  return function(...args) {
    if (wait) {
      storedArgs = args;
      storedThis = this;
      return;
    }

    fn.apply(this, args);
    wait = true;
    setTimeout(checkStoredArgs, limit);
  };
}

module.exports = { throttle };
```
