```javascript
function throttle(fn, limit) {
  let inThrottle = false;
  let lastArgs = null;
  let lastContext = null;

  return function wrapper(...args) {
    if (!inThrottle) {
      fn.apply(this, args);
      inThrottle = true;
      setTimeout(() => {
        inThrottle = false;
        if (lastArgs) {
          const argsToRun = lastArgs;
          const contextToRun = lastContext;
          lastArgs = null;
          lastContext = null;
          wrapper.apply(contextToRun, argsToRun);
        }
      }, limit);
    } else {
      lastArgs = args;
      lastContext = this;
    }
  };
}

module.exports = { throttle };
```
