```javascript
function groupBy(array, fnOrKey) {
  const result = {};
  const getKey = typeof fnOrKey === 'function' ? fnOrKey : (item) => item[fnOrKey];

  for (const item of array) {
    const key = getKey(item);
    if (!result[key]) {
      result[key] = [];
    }
    result[key].push(item);
  }

  return result;
}

module.exports = { groupBy };
```
