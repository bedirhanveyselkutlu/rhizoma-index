```python
import re

def count_words(text: str) -> dict[str, int]:
    # Replace non-alphanumeric characters with spaces, convert to lowercase, and split into words
    cleaned = re.sub(r'[^a-zA-Z0-9]', ' ', text).lower()
    words = cleaned.split()
    
    # Count occurrences
    counts = {}
    for w in words:
        counts[w] = counts.get(w, 0) + 1
    return counts
```
