def is_valid_parentheses(s: str) -> bool:
    stack = []
    mapping = {')': '(', '}': '{', ']': '['}
    for ch in s:
        if ch in mapping.values():          # opening bracket
            stack.append(ch)
        elif ch in mapping:                 # closing bracket
            if not stack or stack.pop() != mapping[ch]:
                return False
        else:
            # If characters other than brackets appear, ignore or return False
            return False
    return not stack
