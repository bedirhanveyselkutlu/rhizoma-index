```javascript
function throttle(fn, limit) {
  let lastCall = 0;
  let timeout = null;

  return function (...args) {
    const now = Date.now();
    const remaining = limit - (now - lastCall);
    const context = this;

    if (remaining <= 0) {
      if (timeout) {
        clearTimeout(timeout);
        timeout = null;
      }
      lastCall = now;
      fn.apply(context, args);
    } else if (!timeout) {
      timeout = setTimeout(() => {
        timeout = null;
        lastCall = Date.now();
        fn.apply(context, args);
      }, remaining);
    }
  };
}

module.exports = { throttle };
```
