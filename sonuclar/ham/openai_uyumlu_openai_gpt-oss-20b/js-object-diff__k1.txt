```javascript
function objectDiff(obj1, obj2) {
  const diff = {};
  const keys = new Set([...Object.keys(obj1), ...Object.keys(obj2)]);
  for (const key of keys) {
    const v1 = obj1[key];
    const v2 = obj2[key];
    if (v1 === v2) continue;
    diff[key] = [v1, v2];
  }
  return diff;
}

module.exports = { objectDiff };
```
