===== HAM YANIT =====
This is the classic LeetCode 2694 Event Emitter problem. `emit` should return an array of callback results in subscription order; unsubscribing removes only that specific subscription (even if the same callback was subscribed multiple times).

```javascript
class EventEmitter {
  constructor() {
    this.events = new Map();
  }

  subscribe(eventName, callback) {
    if (!this.events.has(eventName)) {
      this.events.set(eventName, []);
    }
    const listeners = this.events.get(eventName);
    const entry = { callback };
    listeners.push(entry);

    let active = true;
    return {
      unsubscribe: () => {
        if (!active) return;
        active = false;
        const list = this.events.get(eventName);
        if (!list) return;
        const idx = list.indexOf(entry);
        if (idx !== -1) list.splice(idx, 1);
        if (list.length === 0) this.events.delete(eventName);
      }
    };
  }

  emit(eventName, args = []) {
    const listeners = this.events.get(eventName);
    if (!listeners || listeners.length === 0) return [];
    const snapshot = listeners.slice();
    const results = [];
    for (const entry of snapshot) {
      results.push(entry.callback(...args));
    }
    return results;
  }
}

module.exports = { EventEmitter };
```
