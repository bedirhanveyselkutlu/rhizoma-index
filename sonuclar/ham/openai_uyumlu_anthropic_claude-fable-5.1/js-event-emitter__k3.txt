This is the LeetCode "Event Emitter" problem. `emit` should return an array of results from callbacks, in subscription order, with `args` defaulting to `[]`.

```javascript
class EventEmitter {
  constructor() {
    this.events = new Map();
  }

  subscribe(eventName, callback) {
    if (!this.events.has(eventName)) {
      this.events.set(eventName, []);
    }
    const listeners = this.events.get(eventName);
    const entry = { callback };
    listeners.push(entry);

    return {
      unsubscribe: () => {
        const list = this.events.get(eventName);
        if (!list) return;
        const idx = list.indexOf(entry);
        if (idx !== -1) {
          list.splice(idx, 1);
        }
        if (list.length === 0) {
          this.events.delete(eventName);
        }
      },
    };
  }

  emit(eventName, args = []) {
    const listeners = this.events.get(eventName);
    if (!listeners || listeners.length === 0) {
      return [];
    }
    const snapshot = listeners.slice();
    return snapshot.map((entry) => entry.callback(...args));
  }
}

module.exports = { EventEmitter };
```
