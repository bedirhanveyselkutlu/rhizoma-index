function throttle(fn, limit) {
  let lastCall = 0;
  let timeout = null;
  let lastArgs = null;
  let lastThis = null;

  return function (...args) {
    const now = Date.now();
    const remaining = limit - (now - lastCall);

    if (remaining <= 0 || remaining > limit) {
      if (timeout) {
        clearTimeout(timeout);
        timeout = null;
      }
      lastCall = now;
      return fn.apply(this, args);
    } else {
      lastArgs = args;
      lastThis = this;
      if (!timeout) {
        timeout = setTimeout(() => {
          lastCall = Date.now();
          timeout = null;
          fn.apply(lastThis, lastArgs);
          lastArgs = null;
          lastThis = null;
        }, remaining);
      }
    }
  };
}

module.exports = { throttle };
