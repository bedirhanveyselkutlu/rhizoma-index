```python
import re

def count_words(text: str) -> dict[str, int]:
    counts: dict[str, int] = {}
    for word in re.findall(r"[A-Za-z0-9']+", text.lower()):
        word = word.strip("'")
        if word:
            counts[word] = counts.get(word, 0) + 1
    return counts
```
