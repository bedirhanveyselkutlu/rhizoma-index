import heapq


class MedianFinder:
    def __init__(self):
        self._lower = []
        self._upper = []

    def add_num(self, num: int) -> None:
        heapq.heappush(self._upper, -heapq.heappushpop(self._lower, -num))
        if len(self._upper) > len(self._lower):
            heapq.heappush(self._lower, -heapq.heappop(self._upper))

    def find_median(self) -> float:
        if not self._lower:
            raise ValueError("No numbers have been added")
        if len(self._lower) > len(self._upper):
            return float(-self._lower[0])
        return (-self._lower[0] + self._upper[0]) / 2.0
