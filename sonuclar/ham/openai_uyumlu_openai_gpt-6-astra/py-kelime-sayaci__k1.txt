```python
import string


def count_words(text: str) -> dict[str, int]:
    text = text.lower().translate(str.maketrans("", "", string.punctuation))
    counts: dict[str, int] = {}
    for word in text.split():
        counts[word] = counts.get(word, 0) + 1
    return counts
```
