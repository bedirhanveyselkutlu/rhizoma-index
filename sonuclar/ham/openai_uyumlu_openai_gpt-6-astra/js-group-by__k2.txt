```javascript
function groupBy(array, fnOrKey) {
  const groups = {};
  const selector = typeof fnOrKey === "function"
    ? fnOrKey
    : item => item[fnOrKey];

  array.forEach((item, index) => {
    const key = selector(item, index, array);

    if (!Object.prototype.hasOwnProperty.call(groups, key)) {
      Object.defineProperty(groups, key, {
        value: [],
        enumerable: true,
        configurable: true,
        writable: true
      });
    }

    groups[key].push(item);
  });

  return groups;
}

module.exports = { groupBy };
```
