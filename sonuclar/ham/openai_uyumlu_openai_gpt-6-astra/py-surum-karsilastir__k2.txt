def compare_version(v1: str, v2: str) -> int:
    from itertools import zip_longest

    for a, b in zip_longest(v1.split("."), v2.split("."), fillvalue="0"):
        a = a.lstrip("0") or "0"
        b = b.lstrip("0") or "0"

        if len(a) != len(b):
            return 1 if len(a) > len(b) else -1
        if a != b:
            return 1 if a > b else -1

    return 0
