function levelOrder(root) {
  if (root == null) return [];

  const result = [];
  const queue = [root];
  let head = 0;

  while (head < queue.length) {
    const end = queue.length;
    const level = [];

    while (head < end) {
      const node = queue[head++];
      level.push(node.val);

      for (const child of node.children || []) {
        if (child != null) queue.push(child);
      }
    }

    result.push(level);
  }

  return result;
}

module.exports = { levelOrder };
