function objectDiff(obj1, obj2) {
  const hasOwn = (obj, key) =>
    Object.prototype.hasOwnProperty.call(obj, key);

  function equal(a, b, seen = new WeakMap()) {
    if (Object.is(a, b)) return true;
    if (
      a === null || b === null ||
      typeof a !== "object" || typeof b !== "object"
    ) {
      return false;
    }

    if (Array.isArray(a) !== Array.isArray(b)) return false;
    if (Array.isArray(a) && a.length !== b.length) return false;

    if (a instanceof Date || b instanceof Date) {
      return a instanceof Date && b instanceof Date &&
        Object.is(a.getTime(), b.getTime());
    }

    if (a instanceof RegExp || b instanceof RegExp) {
      return a instanceof RegExp && b instanceof RegExp &&
        a.source === b.source && a.flags === b.flags;
    }

    let partners = seen.get(a);
    if (partners && partners.has(b)) return true;
    if (!partners) {
      partners = new WeakSet();
      seen.set(a, partners);
    }
    partners.add(b);

    const keysA = Object.keys(a);
    const keysB = Object.keys(b);

    return keysA.length === keysB.length &&
      keysA.every(key => hasOwn(b, key) && equal(a[key], b[key], seen));
  }

  const result = {};
  const keys = new Set([...Object.keys(obj1), ...Object.keys(obj2)]);

  for (const key of keys) {
    const inFirst = hasOwn(obj1, key);
    const inSecond = hasOwn(obj2, key);
    const oldValue = inFirst ? obj1[key] : undefined;
    const newValue = inSecond ? obj2[key] : undefined;

    if (inFirst !== inSecond || !equal(oldValue, newValue)) {
      Object.defineProperty(result, key, {
        value: [oldValue, newValue],
        enumerable: true,
        configurable: true,
        writable: true
      });
    }
  }

  return result;
}

module.exports = { objectDiff };
