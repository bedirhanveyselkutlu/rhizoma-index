function throttle(fn, limit) {
  let timerId = null;
  let lastArgs = null;
  let lastThis = null;

  return function (...args) {
    if (timerId === null) {
      fn.apply(this, args);
      timerId = setTimeout(function helper() {
        if (lastArgs) {
          fn.apply(lastThis, lastArgs);
          lastArgs = null;
          lastThis = null;
          timerId = setTimeout(helper, limit);
        } else {
          timerId = null;
        }
      }, limit);
    } else {
      lastArgs = args;
      lastThis = this;
    }
  };
}

module.exports = { throttle };
