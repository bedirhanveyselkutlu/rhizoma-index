```python
from itertools import zip_longest


def compare_version(v1: str, v2: str) -> int:
    parts1 = [int(x) for x in v1.split(".")]
    parts2 = [int(x) for x in v2.split(".")]

    for p1, p2 in zip_longest(parts1, parts2, fillvalue=0):
        if p1 > p2:
            return 1
        elif p1 < p2:
            return -1

    return 0
```
