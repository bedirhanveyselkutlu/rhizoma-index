import heapq

class MedianFinder:
    def __init__(self):
        # max heap for the lower half (store negatives)
        self.lower = []
        # min heap for the upper half
        self.upper = []

    def add_num(self, num: int) -> None:
        # Always push to lower first (as negative for max heap)
        if not self.lower or num <= -self.lower[0]:
            heapq.heappush(self.lower, -num)
        else:
            heapq.heappush(self.upper, num)

        # Rebalance: lower can have at most 1 more element than upper
        if len(self.lower) > len(self.upper) + 1:
            val = -heapq.heappop(self.lower)
            heapq.heappush(self.upper, val)
        elif len(self.upper) > len(self.lower):
            val = heapq.heappop(self.upper)
            heapq.heappush(self.lower, -val)

    def find_median(self) -> float:
        if not self.lower and not self.upper:
            return 0.0
        if len(self.lower) == len(self.upper):
            return (-self.lower[0] + self.upper[0]) / 2.0
        elif len(self.lower) > len(self.upper):
            return float(-self.lower[0])
        else:
            return float(self.upper[0])
