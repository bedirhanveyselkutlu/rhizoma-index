```javascript
function throttle(fn, limit) {
  let lastTime = 0;
  let timerId = null;

  function throttled(...args) {
    const now = Date.now();
    const remainingLimit = limit - (now - lastTime);

    if (remainingLimit <= 0) {
      if (timerId) {
        clearTimeout(timerId);
        timerId = null;
      }
      lastTime = now;
      fn.apply(this, args);
    } else if (!timerId) {
      timerId = setTimeout(() => {
        lastTime = Date.now();
        timerId = null;
        fn.apply(this, args);
      }, remainingLimit);
    }
  }

  return throttled;
}

module.exports = { throttle };
```
