function jsonStringify(value) {
  if (value === null) return 'null';
  if (typeof value === 'boolean') return value ? 'true' : 'false';
  if (typeof value === 'number') {
    if (isNaN(value)) return 'null';
    if (value === Infinity) return 'null';
    if (value === -Infinity) return 'null';
    return String(value);
  }
  if (typeof value === 'string') {
    return '"' + value.replace(/["\\\b\f\n\r\t]/g, (ch) => {
      switch (ch) {
        case '"': return '\\"';
        case '\\': return '\\\\';
        case '\b': return '\\b';
        case '\f': return '\\f';
        case '\n': return '\\n';
        case '\r': return '\\r';
        case '\t': return '\\t';
        default: return ch;
      }
    }) + '"';
  }
  if (Array.isArray(value)) {
    const items = [];
    for (let i = 0; i < value.length; i++) {
      if (value[i] === undefined || typeof value[i] === 'function' || (typeof value[i] === 'symbol')) {
        items.push('null');
      } else {
        items.push(jsonStringify(value[i]));
      }
    }
    return '[' + items.join(',') + ']';
  }
  if (typeof value === 'object') {
    const keys = Object.keys(value);
    const items = [];
    for (let i = 0; i < keys.length; i++) {
      const key = keys[i];
      const val = value[key];
      if (val === undefined || typeof val === 'function' || typeof val === 'symbol') {
        continue;
      }
      items.push(jsonStringify(key) + ':' + jsonStringify(val));
    }
    return '{' + items.join(',') + '}';
  }
  return undefined;
}

module.exports = { jsonStringify };
