function throttle(fn, limit) {
  let previous = 0;
  let timeoutId = null;

  return function (...args) {
    const now = Date.now();
    const remaining = limit - (now - previous);

    if (remaining <= 0) {
      // Execute immediately
      previous = now;
      fn.apply(this, args);
    } else {
      // Schedule execution after the limit
      if (timeoutId) {
        clearTimeout(timeoutId);
      }
      timeoutId = setTimeout(() => {
        previous = Date.now();
        timeoutId = null;
        fn.apply(this, args);
      }, remaining);
    }
  };
}

module.exports = { throttle };
