```python
import string
from collections import Counter

def count_words(text: str) -> dict[str, int]:
    # Remove punctuation
    translator = str.maketrans('', '', string.punctuation)
    cleaned = text.translate(translator).lower()
    # Split into words
    words = cleaned.split()
    # Count occurrences
    return dict(Counter(words))
```
