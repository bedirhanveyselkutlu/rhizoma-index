function objectDiff(obj1, obj2) {
  const diff = {};
  const allKeys = new Set([...Object.keys(obj1), ...Object.keys(obj2)]);

  for (const key of allKeys) {
    if (obj1[key] !== obj2[key]) {
      diff[key] = [obj1[key], obj2[key]];
    }
  }

  return diff;
}

module.exports = { objectDiff };
