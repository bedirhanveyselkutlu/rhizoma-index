function objectDiff(obj1, obj2) {
  const diff = [];
  for (const key in obj1) {
    if (obj1.hasOwnProperty(key) && obj2.hasOwnProperty(key)) {
      if (obj1[key] !== obj2[key]) {
        diff.push([key, obj1[key], obj2[key]]);
      }
    }
  }
  return diff;
}
