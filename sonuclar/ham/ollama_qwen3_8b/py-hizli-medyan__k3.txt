class MedianFinder:
    def __init__(self):
        self.lower = []  # max heap (inverted using negative values)
        self.upper = []  # min heap

    def add_num(self, num: int) -> None:
        import heapq
        if not self.lower or num <= -self.lower[0]:
            heapq.heappush(self.lower, -num)
        else:
            heapq.heappush(self.upper, num)
        
        if len(self.lower) > len(self.upper) + 1:
            val = -heapq.heappop(self.lower)
            heapq.heappush(self.upper, val)
        elif len(self.upper) > len(self.lower):
            val = heapq.heappop(self.upper)
            heapq.heappush(self.lower, -val)

    def find_median(self) -> float:
        if (len(self.lower) + len(self.upper)) % 2 == 1:
            return -self.lower[0]
        else:
            return (-self.lower[0] + self.upper[0]) / 2
