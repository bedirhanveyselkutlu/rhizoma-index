class MedianFinder:
    def __init__(self):
        self.max_heap = []  # Max heap to store the smaller half
        self.min_heap = []  # Min heap to store the larger half

    def add_num(self, num: int) -> None:
        # Add to max_heap first, then balance
        heapq.heappush(self.max_heap, -num)
        heapq.heappush(self.min_heap, -self.max_heap[0])
        heapq.heappop(self.max_heap)
        
        # Balance the heaps
        if len(self.min_heap) > len(self.max_heap):
            heapq.heappush(self.max_heap, -self.min_heap[0])
            heapq.heappop(self.min_heap)

    def find_median(self) -> float:
        if len(self.min_heap) == len(self.max_heap):
            return (-self.max_heap[0] + self.min_heap[0]) / 2
        else:
            return -self.max_heap[0]
