```javascript
function groupBy(array, fnOrKey) {
  return array.reduce((acc, curr) => {
    const key = typeof fnOrKey === 'function' ? fnOrKey(curr) : curr[fnOrKey];
    if (!acc[key]) acc[key] = [];
    acc[key].push(curr);
    return acc;
  }, {});
}

module.exports = { groupBy };
```
