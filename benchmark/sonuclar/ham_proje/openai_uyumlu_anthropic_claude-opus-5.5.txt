```python
import logging
import re
from decimal import Decimal, ROUND_HALF_UP, localcontext

audit_logger = logging.getLogger("billing.audit")

# ---------------------------------------------------------------- Task A
# [0-9] rather than \d, because \d also matches non-ASCII digits such as "٣".
_DURATION_RE = re.compile(r"(?:([0-9]+)h)?(?:([0-9]+)m)?(?:([0-9]+)s)?")


def parse_duration(text: str) -> int:
    """Parse strings like '2h', '45m', '1h5m10s' into total seconds."""
    if not isinstance(text, str):
        raise ValueError(f"duration must be a string, got {type(text).__name__}")
    match = _DURATION_RE.fullmatch(text)
    # The empty string matches the regex with all groups None, so reject that too.
    if match is None or all(g is None for g in match.groups()):
        raise ValueError(f"invalid duration: {text!r}")
    hours, minutes, seconds = (int(g) if g is not None else 0 for g in match.groups())
    return hours * 3600 + minutes * 60 + seconds


# ---------------------------------------------------------------- Task B (+ D, E hook)
def apply_discount(total_cents: int, percent, *, max_percent=100) -> int:
    """Return total_cents discounted by `percent`, rounded to whole cents, halves up."""
    if isinstance(total_cents, bool) or not isinstance(total_cents, int):
        raise TypeError("total_cents must be an int")
    if isinstance(percent, bool) or not isinstance(percent, (int, float, Decimal)):
        raise TypeError("percent must be a number")

    # str() first so that e.g. 10.1 means Decimal('10.1'), not its binary approximation.
    p = percent if isinstance(percent, Decimal) else Decimal(str(percent))
    limit = Decimal(str(max_percent))

    if not p.is_finite() or p < 0 or p > 100:        # NaN/inf are rejected here
        raise ValueError(f"percent must be between 0 and 100, got {percent!r}")
    if not limit.is_finite() or limit < 0 or limit > 100:
        raise ValueError(f"max_percent must be between 0 and 100, got {max_percent!r}")
    if p > limit:
        raise ValueError(f"percent {percent!r} exceeds maximum allowed discount {max_percent!r}")

    with localcontext() as ctx:
        ctx.prec = 60  # enough headroom that the multiplication is never rounded early
        discounted = Decimal(total_cents) * (100 - p) / 100
        result = int(discounted.quantize(Decimal(1), rounding=ROUND_HALF_UP))

    # Task D: see note below; logging.audit() does not exist.
    audit_logger.info(
        "discount applied: total_cents=%d percent=%s result_cents=%d",
        total_cents, p, result,
    )
    return result


# ---------------------------------------------------------------- Task C
MAX_AMOUNT_CENTS = 1_000_000_000  # 10,000,000.00 TRY


def format_amount(cents: int) -> str:
    """Format cents as lira, e.g. 1234 -> '12.34 TRY'. Width is NOT fixed (see note)."""
    if isinstance(cents, bool) or not isinstance(cents, int):
        raise TypeError("cents must be an int")
    if abs(cents) > MAX_AMOUNT_CENTS:
        raise ValueError(f"amount exceeds 10,000,000.00 TRY: {cents}")
    sign = "-" if cents < 0 else ""
    lira, kurus = divmod(abs(cents), 100)
    return f"{sign}{lira}.{kurus:02d} TRY"
```

## Notes on each task

**Task A.** Each unit is optional, but the order h, m, s is enforced by the regex, and at least one unit must be present. Inputs like `"2h"`, `"1h30m"`, `"0s"` and `"1h5m10s"` work. Inputs like `""`, `"30m1h"`, `"1h1h"`, `"1h 30m"`, `"-5m"`, `"1.5h"` and `"h"` raise `ValueError`. Leading zeros such as `"05m"` are accepted. Tell me if they shouldn't be.

**Task B.** The original code had two problems:

1. `int()` truncates instead of rounding. For example, 500.9 became 500.
2. Float arithmetic can land just below a true .5.

The fix uses `Decimal` with `ROUND_HALF_UP`, so 1001 cents at 50% gives 501. Out-of-range values, NaN and infinity raise `ValueError`. Non-numeric inputs raise `TypeError`; switch these to `ValueError` if you prefer a single exception type.

One caveat on negative totals such as refunds: `ROUND_HALF_UP` rounds away from zero, so -500.5 becomes -501. Tell me if you want -500 instead.
