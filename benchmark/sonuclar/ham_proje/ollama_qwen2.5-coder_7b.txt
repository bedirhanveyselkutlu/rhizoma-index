```python
import re
def parse_duration(text: str) -> int:
    pattern = re.compile(r'(\d+)([hms])')
    matches = pattern.findall(text)
    total_seconds = 0
    for value, unit in matches:
        value = int(value)
        if unit == 'h':
            total_seconds += value * 3600
        elif unit == 'm':
            total_seconds += value * 60
        elif unit == 's':
            total_seconds += value
    return total_seconds
```
