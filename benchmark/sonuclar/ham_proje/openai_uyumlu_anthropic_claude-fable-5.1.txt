import logging
import re

# [0-9] rather than \d so Unicode digits like "٣" are rejected.
_DURATION_RE = re.compile(r"(?:([0-9]+)h)?(?:([0-9]+)m)?(?:([0-9]+)s)?")


def parse_duration(text: str) -> int:
    """Parse "2h", "45m", "90s", "1h30m", "1h5m10s" -> total seconds.

    Units must be in h, m, s order, each at most once, each preceded by a
    non-negative integer. Anything else (including "") raises ValueError.
    """
    if not isinstance(text, str):
        raise ValueError(f"duration must be a str, got {type(text).__name__}")
    m = _DURATION_RE.fullmatch(text)
    # all-None groups means the (valid) empty match -> reject
    if m is None or not any(g is not None for g in m.groups()):
        raise ValueError(f"invalid duration: {text!r}")
    h, mi, s = (int(g) if g is not None else 0 for g in m.groups())
    return h * 3600 + mi * 60 + s

def apply_discount(total_cents: int, percent: Real) -> int:
    """Return the discounted amount in whole cents, rounded half-up.

    percent must be within [0, 100] (and <= MAX_DISCOUNT_PERCENT if set),
    otherwise ValueError.
    """
    if isinstance(total_cents, bool) or not isinstance(total_cents, int):
        raise TypeError("total_cents must be an int number of cents")
    if isinstance(percent, bool) or not isinstance(percent, (Real, Decimal)):
        raise TypeError("percent must be a number")
    if not (0 <= percent <= 100):  # also rejects NaN
        raise ValueError(f"percent must be between 0 and 100, got {percent!r}")

    pct = Decimal(str(percent))  # str() avoids binary-float artefacts
    if MAX_DISCOUNT_PERCENT is not None and pct > MAX_DISCOUNT_PERCENT:
        raise ValueError(
            f"discount {pct}% exceeds maximum allowed {MAX_DISCOUNT_PERCENT}%"
        )

    discounted = (Decimal(total_cents) * (Decimal(100) - pct) / Decimal(100))
    result = int(discounted.quantize(Decimal(1), rounding=ROUND_HALF_UP))

    _audit_log.info(
        "discount applied",
        extra={"total_cents": total_cents, "percent": str(pct), "result_cents": result},
    )
    return result

def format_amount(cents: int) -> str:
    """Format cents as lira with two decimals, e.g. 1234 -> "12.34 TRY"."""
    if isinstance(cents, bool) or not isinstance(cents, int):
        raise TypeError("cents must be an int")
    sign = "-" if cents < 0 else ""
    lira, rem = divmod(abs(cents), 100)
    return f"{sign}{lira}.{rem:02d} TRY"

### Notes on what was fixed in `apply_discount` (Task B)
- Original used `int(...)`, which truncates toward zero (500.5 → 500). Now uses `Decimal` with `ROUND_HALF_UP`, so 500.5 → 501.
- Original did no range check; now raises `ValueError` outside [0, 100] (NaN is also rejected).
- Float `percent` values like `12.5` are converted via `str()` so `12.5` doesn't become `12.4999…`.
- For negative `total_cents`, `ROUND_HALF_UP` rounds halves away from zero (−500.5 → −501). Tell me if you need a different convention for refunds/credits.
