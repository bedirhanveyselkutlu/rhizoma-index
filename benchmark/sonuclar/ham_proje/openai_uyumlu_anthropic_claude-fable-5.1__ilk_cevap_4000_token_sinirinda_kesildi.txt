from __future__ import annotations

import logging
import re
from decimal import Decimal, ROUND_HALF_UP

# ---------------------------------------------------------------------------
# Task A: parse_duration

# [0-9] instead of \d so Unicode digits (e.g. "٣h") are rejected.
# fullmatch instead of ^...$ so a trailing newline is rejected.
_DURATION_RE = re.compile(r"(?:([0-9]+)h)?(?:([0-9]+)m)?(?:([0-9]+)s)?")


def parse_duration(text: str) -> int:
    """Parse "2h", "45m", "90s", "1h30m", "1h5m10s" -> total seconds.

    Units h, m, s; each at most once, in that order; each preceded by a
    non-negative integer. Anything else (including "") raises ValueError.
    """
    if not isinstance(text, str):
        raise ValueError(f"duration must be a str, got {type(text).__name__}")
    m = _DURATION_RE.fullmatch(text)
    # The regex also matches "" (all groups optional), so require >= 1 unit.
    if m is None or not any(g is not None for g in m.groups()):
        raise ValueError(f"invalid duration: {text!r}")
    h, mi, s = (int(g) if g is not None else 0 for g in m.groups())
    return h * 3600 + mi * 60 + s

def apply_discount(total_cents: int, percent, *, max_percent=None) -> int:
    """Return the discounted amount in whole cents, rounded half-up.

    percent must be a number in [0, 100]; otherwise ValueError.
    If max_percent is given (e.g. from pricing_rules.yaml) and percent
    exceeds it, ValueError is raised.
    """
    if isinstance(percent, bool) or not isinstance(percent, (int, float, Decimal)):
        raise ValueError(f"percent must be a number, got {percent!r}")
    # `not (0 <= p <= 100)` also rejects NaN, since comparisons with NaN are False.
    if not (0 <= percent <= 100):
        raise ValueError(f"percent must be between 0 and 100, got {percent!r}")
    if max_percent is not None and Decimal(str(percent)) > Decimal(str(max_percent)):
        raise ValueError(f"percent {percent!r} exceeds maximum allowed {max_percent!r}")

    # Decimal(str(...)) avoids binary-float artefacts (e.g. 0.1 * 3).
    pct = Decimal(str(percent))
    amount = Decimal(total_cents) * (Decimal(100) - pct) / Decimal(100)
    result = int(amount.quantize(Decimal("1"), rounding=ROUND_HALF_UP))

    _audit("discount applied: total_cents=%s percent=%s result_cents=%s",
           total_cents, percent, result)
    return result

def format_amount(cents: int) -> str:
    """Format cents as lira with two decimals, e.g. 1234 -> "12.34 TRY"."""
    if not isinstance(cents, int) or isinstance(cents, bool):
        raise ValueError(f"cents must be an int, got {cents!r}")
    sign = "-" if cents < 0 else ""
    lira, rem = divmod(abs(cents), 100)
    return f"{sign}{lira}.{rem:02d} TRY"
