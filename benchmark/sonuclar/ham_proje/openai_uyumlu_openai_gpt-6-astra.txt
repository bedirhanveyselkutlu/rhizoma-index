## Implementation

```python
import logging
import re
from fractions import Fraction

_DURATION_RE = re.compile(
    r"(?:(?P<h>[0-9]+)h)?"
    r"(?:(?P<m>[0-9]+)m)?"
    r"(?:(?P<s>[0-9]+)s)?"
)

audit_logger = logging.getLogger("billing.audit")


def parse_duration(text: str) -> int:
    if not isinstance(text, str) or not text:
        raise ValueError("Duration must be a nonempty string")

    match = _DURATION_RE.fullmatch(text)
    if match is None:
        raise ValueError("Invalid duration; expected ordered h, m, s units")

    return sum(
        int(match.group(unit) or "0") * multiplier
        for unit, multiplier in (("h", 3600), ("m", 60), ("s", 1))
    )


def _validate_percent(value, name: str) -> Fraction:
    try:
        # Decimal spelling avoids introducing binary floating-point
        # artifacts. Also supports Decimal and Fraction values.
        result = Fraction(str(value))
    except (ValueError, ZeroDivisionError, OverflowError) as exc:
        raise ValueError(
            f"{name} must be a finite number between 0 and 100"
        ) from exc

    if not 0 <= result <= 100:
        raise ValueError(f"{name} must be between 0 and 100")
    return result


def _round_half_up(value: Fraction) -> int:
    magnitude, remainder = divmod(abs(value.numerator), value.denominator)
    if 2 * remainder >= value.denominator:
        magnitude += 1
    return -magnitude if value < 0 else magnitude


def apply_discount(total_cents: int, percent, *, max_percent=100) -> int:
    """
    Apply a discount and round to whole cents, with ties away from zero.

    Production callers must supply max_percent from validated pricing rules.
    The default only enforces the general 0–100 constraint.
    """
    if type(total_cents) is not int:
        raise TypeError("total_cents must be an integer")

    discount = _validate_percent(percent, "percent")
    maximum = _validate_percent(max_percent, "max_percent")

    if discount > maximum:
        raise ValueError("percent exceeds the configured maximum discount")

    exact_cents = Fraction(total_cents) * (100 - discount) / 100
    discounted_cents = _round_half_up(exact_cents)

    audit_logger.info(
        "discount_applied",
        extra={
            "total_cents": total_cents,
            "discount_percent": str(discount),
            "maximum_discount_percent": str(maximum),
            "discounted_cents": discounted_cents,
        },
    )

    return discounted_cents


def format_amount(cents: int) -> str:
    """
    Proposed resolution of Task C: variable-width output, always two decimals.
    Supports nonnegative amounts through 10,000,000.00 TRY.
    """
    if type(cents) is not int:
        raise TypeError("cents must be an integer")
    if not 0 <= cents <= 1_000_000_000:
        raise ValueError("Amount must be between 0 and 10,000,000.00 TRY")

    whole, fractional = divmod(cents, 100)
    return f"{whole}.{fractional:02d} TRY"
```
